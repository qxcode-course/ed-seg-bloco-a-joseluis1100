func carpete(pen *Pen, tam, depth float64) {
	if depth == 0 {
		return
	}
	x, y := pen.x, pen.y
	pen.Up()
	pen.SetPosition(x-tam/2, y-tam/2)
	pen.Down()
	pen.FillSquare(tam, tam)
	for i := range 3 {
		for j := range 3 {
			pen.Up()
			pen.SetPosition(x+(float64(i-1)*tam), y+(float64(j-1)*tam))
			carpete(pen, tam/3, depth-1)
		}
	}
}

	fmt.Println("PNG file created successfully.")
}
